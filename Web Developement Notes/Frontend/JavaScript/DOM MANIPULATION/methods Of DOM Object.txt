 document.getElementById('title') -> gives the element with id "title"
 document.getElementById('title').id -> gives the element's id
 document.getElementById('title').className -> gives the class name
 document.getElementById('title').getAttribute('id') -> gets the id attribute
 document.getElementById('title').getAttribute('class') -> gets the class attribute
 document.getElementById('title').getAttribute('class', 'test')

const title = document.getElementById('title');

title.style.backgroundColor = 'green';
title.style.padding = '15px';
title.style.borderRadius = '15px';

title.textContent; ->    gives all the content, even hidden text
title.innerHTML; ->   gives content along with HTML
title.innerText;   ->      gives only visible text

 document.getElementsByClassName('heading') -> gives all elements with class "heading"

 document.querySelector('h2') -> gives the first h2
 document.querySelector('#title') -> gives the element with id title
 document.querySelector('.heading') -> gives the first element with class heading
 document.querySelector('input[type="password"]') -> selects the password input

const tempList = document.querySelectorAll('li') -> gives the nodelist datatype not arraylist
tempList[0].style.color = 'green'

tempList.foreach(function (l)){
   l.style.backgroundColor = 'green'
}
